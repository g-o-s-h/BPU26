```py
# TODO: 计算校验码
def get_code(prefix: str) -> int:
    digits = [int(prefix[i]) * (i + 1) 
              for i in range(len(prefix))]  # 按公式把每一项乘出来
    return sum(digits) % 11  # 求和，模 11 并返回
```
