```c++
#include <iostream>
#include <string>

int count[26]; // a 0, z 25

bool isPrime(int number) {
    // 2 to n - 1; if (n % i == 0) flag = false; break;
    if (number == 0 || number == 1) {
        return false;
    }

    for (int i = 2; i < number; i ++) {
        if (number % i == 0) {
            return false;
        }
    }
    return true;
}

int main() {
    std::string word;
    std::cin >> word;

    for (auto letter : word) {
        count[letter - 'a'] ++;
    }

    int maximum = -1, minimum = 101;
    for (int i = 0; i < 26; i ++) {
        if (count[i] == 0) continue;
        if (maximum < count[i]) maximum = count[i];
        if (minimum > count[i]) minimum = count[i];
    }
    // std::cerr << "# " << maximum << " " << minimum << std::endl;

    if (isPrime(maximum - minimum)) {
        std::cout << "Lucky Word\n" << maximum - minimum;
    } else {
        std::cout << "No Answer\n0";
    }
}
```

求阶乘 Factorial
```c++
#include <iostream>

int factorial(int number) {
    if (number == 1) return 1;
    return number * factorial(number - 1);
}

int main() {
    int n;
    std::cin >> n;
    std::cout << factorial(n);
}
```
